automacro set_config {
  ConfigKeyNot itemsTakeAuto $loot, attackAuto $attack, route_randomWalk $walk, tankMode $tank, attackLooters $looter
  CheckOnAI auto, manual
  exclusive 1
  priority 1
  call set_config
}

macro set_config {
  if ($.ConfigKeyNotLastKeyValue == none) {
    add_key($.ConfigKeyNotLastKey, $.ConfigKeyNotLastWantedValue)
  } else {
    do conf $.ConfigKeyNotLastKey $.ConfigKeyNotLastWantedValue
  }
}

macro set_eq {
  $head = 18730 #topHead
  $off  = 2112  #leftHand
  $main = 1243  #rightHand
  $robe  = 2510 #robe
  $body = 2352  #armor
  $foot = 2414  #shoes
  if ($conf[0] > 0) {
    $acc2 = 2819
  }
  if ($conf[1] > 0) {
    $acc2 = 2824
  }
  if ($conf[2] > 0) {
    $acc1 = 2821
  } else {
    $acc1 = 2820
  }
}

automacro set_equip {
  IsNotEquippedID topHead $head, leftHand $off, rightHand $main, robe $robe, armor $body, shoes $foot, leftAccessory $acc1, rightAccessory $acc2
  CheckOnAI auto, manual
  exclusive 1
  call {
   log equiping $.IsNotEquippedIDLastSlot
     switch ($.IsNotEquippedIDLastSlot) {
      case (== topHead) {
        do eq topHead &inventory ($head)
      }
      case (== leftHand) {
        do eq leftHand &inventory ($off)
      }
      case (== rightHand) {
        do eq rightHand &inventory ($main)
      }
      case (== robe) {
        do eq robe &inventory ($robe)
      }
      case (== armor) {
        do eq armor &inventory ($body)
      }
      case (== shoes) {
        do eq shoes &inventory ($foot)
      }
      case (== leftAccessory) {
        do eq leftAccessory &inventory ($acc1)
      }
      case (== rightAccessory) {
        do eq rightAccessory &inventory ($acc2)
      }
    }
  }
}

macro set_monster {
  $match1 = &eval ((defined Misc::mon_control("$monster[0]")->{attack_auto}?Misc::mon_control("$monster[0]")->{attack_auto}:"None") eq $monster[1])
  $match2 = &eval ((defined Misc::mon_control("$monster[0]")->{teleport_auto}?Misc::mon_control("$monster[0]")->{teleport_auto}:"None") eq $monster[2])
  $match3 = &eval ((defined Misc::mon_control("$monster[0]")->{teleport_search}?Misc::mon_control("$monster[0]")->{teleport_search}:"None") eq $monster[3])
  if ( !$match1 || !$match2 || !$match3 ) {
    rewrite(@monster)
    do reload mon_control
    pause 2
  }
}

sub replace_string {
  my ($string, $from, $to, $interpolate) = @_;
  return $string if ($from eq "");
  if ($interpolate eq "") {
    $string =~s/$from/$to/ig;
  } else {
    my ($match) = $string =~ m/$from/;
    if ($1 eq "") {
      $string =~ s/$from/$to/;
    } else {
      $regex = '"'. $to . $interpolate . '"';
      $string =~ s/$from/$regex/ee;
    }
  }
  return $string;
}

sub getPos {
  my ($loc, $dx, $dy) = @_;
  my @pos = split ' ', $loc;
  @pos[0] += $dx;
  @pos[1] += $dy;
  return @pos;
}

sub add_key {
    my ($key, $value) = @_;
    configModify($key, $value);
}

sub rewrite {
  my (@mob) = @_;
  my $monster = Misc::mon_control("$mob[0]");
  my @lines = ();
  my $found = 0;
  my $skip = $mob[$#mob] if ($#mob > 3);

  if (open(FILE, "<:utf8", Settings::getControlFilename("mon_control.txt"))) {
    while (<FILE>) {
      s/\x{FEFF}//g; chomp;
      if (/^#/ || /^\n/ || /^\r/) {
        push @lines,$_;
        next
      }
      /^(\d+|([a-zA-Z' ]+)*) -?\d/;
      if ("$mob[0]" eq $1 && defined $monster) {
        $found = 1;
        $_ = $1 . " $mob[1] $mob[2] $mob[3] $monster->{attack_lvl} $monster->{attack_jlvl} $monster->{attack_hp} $monster->{attack_sp} $monster->{weight}";
        s/\s+$//;
        push(@lines, $_) if (!$skip);
      } else {
        push(@lines, $_);
      }
    }
    push(@lines, $mob[0] . " $mob[1] $mob[2] $mob[3]") if (!$found);
    close FILE;
  }
  open(FILE, ">:utf8", Settings::getControlFilename("mon_control.txt"));
  print FILE join "\n", @lines;
  close FILE;
}
